```dart
enum FoodUnit { grams, milliliters }

class Ingredient {
  final String name;
  final int percentage;
  const Ingredient({required this.name, required this.percentage});
  factory Ingredient.fromJson(Map<String, dynamic> json) => Ingredient(name: json['name'] as String? ?? '', percentage: json['percentage'] as int? ?? 0);
  Map<String, dynamic> toJson() => {'name': name, 'percentage': percentage};
  Ingredient copyWith({String? name, int? percentage}) => Ingredient(name: name ?? this.name, percentage: percentage ?? this.percentage);
}

class FoodItem {
  final String id;
  final String name;
  final double calories;
  final double protein;
  final double carbs;
  final double fat;
  final double sugar;
  final double quantity;
  final String? imageUrl;
  final DateTime timestamp;
  final List<Ingredient> ingredients;
  final FoodUnit unit;
  final double? confidenceScore;
  final String? aiModel;

  FoodItem({
    required this.id, required this.name, required this.calories, required this.protein,
    required this.carbs, required this.fat, required this.sugar, required this.quantity,
    this.imageUrl, required this.timestamp, this.ingredients = const [], this.unit = FoodUnit.grams,
    this.confidenceScore, this.aiModel,
  });

  factory FoodItem.fromJson(Map<String, dynamic> json) {
    var ingredientsList = <Ingredient>[];
    if (json['ingredients'] != null) {
      final list = json['ingredients'] as List;
      for (var item in list) {
        if (item is Map<String, dynamic>) ingredientsList.add(Ingredient.fromJson(item));
      }
    }
    return FoodItem(
      id: json['id'] as String? ?? '', name: json['name'] as String? ?? '',
      calories: (json['calories'] as num?)?.toDouble() ?? 0.0,
      protein: (json['protein'] as num?)?.toDouble() ?? 0.0,
      carbs: (json['carbs'] as num?)?.toDouble() ?? 0.0,
      fat: (json['fat'] as num?)?.toDouble() ?? 0.0,
      sugar: (json['sugar'] as num?)?.toDouble() ?? 0.0,
      quantity: (json['quantity'] as num?)?.toDouble() ?? 0.0,
      imageUrl: json['imageUrl'] as String?,
      timestamp: DateTime.parse(json['timestamp'] as String? ?? DateTime.now().toIso8601String()),
      ingredients: ingredientsList,
      unit: json['unit'] == 'ml' ? FoodUnit.milliliters : FoodUnit.grams,
      confidenceScore: (json['confidenceScore'] as num?)?.toDouble(),
      aiModel: json['aiModel'] as String?,
    );
  }

  Map<String, dynamic> toJson() => {
    'id': id, 'name': name, 'calories': calories, 'protein': protein, 'carbs': carbs,
    'fat': fat, 'sugar': sugar, 'quantity': quantity, 'imageUrl': imageUrl,
    'timestamp': timestamp.toIso8601String(), 'ingredients': ingredients.map((i) => i.toJson()).toList(),
    'unit': unit == FoodUnit.milliliters ? 'ml' : 'g', 'confidenceScore': confidenceScore, 'aiModel': aiModel,
  };

  FoodItem copyWith({String? name, double? calories, double? protein, double? carbs, double? fat, double? sugar, double? quantity, String? imageUrl, List<Ingredient>? ingredients, FoodUnit? unit, double? confidenceScore, String? aiModel}) {
    return FoodItem(id: id, name: name ?? this.name, calories: calories ?? this.calories, protein: protein ?? this.protein, carbs: carbs ?? this.carbs, fat: fat ?? this.fat, sugar: sugar ?? this.sugar, quantity: quantity ?? this.quantity, imageUrl: imageUrl ?? this.imageUrl, timestamp: timestamp, ingredients: ingredients ?? this.ingredients, unit: unit ?? this.unit, confidenceScore: confidenceScore ?? this.confidenceScore, aiModel: aiModel ?? this.aiModel);
  }

  double get totalIngredientPercentage => ingredients.isEmpty ? 0 : ingredients.fold(0, (sum, ing) => sum + ing.percentage);
  List<Ingredient> get sortedIngredients { final sorted = List<Ingredient>.from(ingredients); sorted.sort((a, b) => b.percentage.compareTo(a.percentage)); return sorted; }
  String get unitSymbol => unit == FoodUnit.grams ? 'g' : 'ml';
}
```

### 4.1 food_service.dart
```dart
import 'dart:convert';
import 'package:flutter/foundation.dart';
import 'package:shared_preferences/shared_preferences.dart';
import 'ollama_service.dart';
import 'google_ai_service.dart';

class FoodService {
  static final OllamaService _ollamaService = OllamaService();
  static final GoogleAiService _googleService = GoogleAiService();
  static bool _isInitialized = false;
  static const String _providerPref = 'ai_provider_mode';
  static const List<Map<String, String>> availableModels = [];
  static String get model => _activeProvider == 'google' ? _googleService.selectedModel : _ollamaService.selectedModel;
  static String? get apiKey => _activeProvider == 'google' ? _googleService.apiKey : 'ollama';
  static String _activeProvider = 'ollama';
  static String get activeProvider => _activeProvider;

  static Future<void> initFromPrefs() async {
    if (_isInitialized) return;
    try {
      final prefs = await SharedPreferences.getInstance();
      _activeProvider = prefs.getString(_providerPref) ?? 'ollama';
      await _ollamaService.initialize();
      await _googleService.initialize();
      _isInitialized = true;
    } catch (e) {
      _isInitialized = true;
      debugPrint('FoodService: Error inicializando: $e');
    }
  }

  static Future<void> setProvider(String provider) async {
    _activeProvider = provider;
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(_providerPref, provider);
  }

  static void setModel(String m) {
    if (_activeProvider == 'google') { _googleService.updateModel(m); } else { _ollamaService.updateModel(m); }
  }

  static void setApiKey(String k) {
    if (_activeProvider == 'google') { _googleService.updateApiKey(k); }
  }

  static Future<FoodAnalysisResult> analyzeFoodImageFromBytes(Uint8List imageBytes) async {
    if (!_isInitialized) await initFromPrefs();
    try {
      final base64Image = base64Encode(imageBytes);
      final prompt = 'Analiza esta imagen de comida. Responde SOLO en JSON valido: {"foods":["nombre1"], "estimatedCalories":500, "macros":{"protein":25,"carbs":50,"fat":15}, "confidence":"high"}';
      String responseText;
      bool isSuccess;
      String? error;

      if (_activeProvider == 'google') {
        final resp = await _googleService.generateResponseWithImage(prompt: prompt, imageBase64: base64Image);
        isSuccess = resp.isSuccess; responseText = resp.text; error = resp.error;
      } else {
        final resp = await _ollamaService.generateResponseWithImage(prompt: prompt, model: _ollamaService.selectedModel, imageBase64: base64Image);
        isSuccess = resp.isSuccess; responseText = resp.text; error = resp.error;
      }

      if (isSuccess) {
        try {
          final jsonMatch = RegExp(r'\{[\s\S]*\}').firstMatch(responseText);
          if (jsonMatch != null) {
            final jsonData = jsonDecode(jsonMatch.group(0)!);
            return FoodAnalysisResult.fromJson(jsonData);
          }
        } catch (parseErr) {
          debugPrint('FoodService: Error parseando JSON: $parseErr');
        }
        return FoodAnalysisResult(foods: ['Comida detectada'], estimatedCalories: 300, protein: 20, carbs: 30, fat: 10, confidence: 'low');
      } else {
        return FoodAnalysisResult.error(error ?? 'Error desconocido');
      }
    } catch (e) {
      return FoodAnalysisResult.error('Error: $e');
    }
  }

  static Future<FoodAnalysisResult> estimateCaloriesFromText(String description) async {
    if (!_isInitialized) await initFromPrefs();
    try {
      final prompt = 'Comida: "$description". Responde SOLO en JSON: {"foods":["nombre"], "estimatedCalories":400, "macros":{"protein":20,"carbs":40,"fat":15}, "confidence":"medium"}';
      String responseText;
      bool isSuccess;
      String? error;

      if (_activeProvider == 'google') {
        final resp = await _googleService.generateResponse(prompt: prompt);
        isSuccess = resp.isSuccess; responseText = resp.text; error = resp.error;
      } else {
        final resp = await _ollamaService.generateResponse(prompt: prompt, model: _ollamaService.selectedModel);
        isSuccess = resp.isSuccess; responseText = resp.text; error = resp.error;
      }

      if (isSuccess) {
        try {
          final jsonMatch = RegExp(r'\{[\s\S]*\}').firstMatch(responseText);
          if (jsonMatch != null) return FoodAnalysisResult.fromJson(jsonDecode(jsonMatch.group(0)!));
        } catch (parseErr) { debugPrint('Error: $parseErr'); }
        return FoodAnalysisResult(foods: [description], estimatedCalories: 250, protein: 15, carbs: 25, fat: 8, confidence: 'low');
      } else {
        return FoodAnalysisResult.error(error ?? 'Error');
      }
    } catch (e) {
      return FoodAnalysisResult.error('Error: $e');
    }
  }

  static Future<String> chatCompletion(String userMessage, {List<String>? context}) async {
    if (!_isInitialized) await initFromPrefs();
    try {
      final ctxStr = context != null && context.isNotEmpty ? 'Historial:\n${context.join("\n")}\n\n' : '';
      final prompt = 'Eres asesor nutricional experto. Responde en español.\n${ctxStr}Usuario: $userMessage';
      if (_activeProvider == 'google') {
        final resp = await _googleService.generateResponse(prompt: prompt);
        return resp.isSuccess ? resp.text : 'Error Google: ${resp.error}';
      } else {
        final resp = await _ollamaService.generateResponse(prompt: prompt, model: _ollamaService.selectedModel);
        return resp.isSuccess ? resp.text : 'Error Ollama: ${resp.error}';
      }
    } catch (e) { return 'Error: $e'; }
```

```dart
abstract class AiModelAdapter {
  String get name;
  Future<AiResult> generateText(String prompt, {Map<String, dynamic>? context});
  Future<AiResult> generateWithImage(String prompt, String imageBase64);
  Future<bool> isAvailable();
}

class AiResult {
  final String text;
  final bool isSuccess;
  final String? error;
  final String? modelUsed;
  final double? confidence;

  AiResult({required this.text, required this.isSuccess, this.error, this.modelUsed, this.confidence});
}

class AiOrchestrator {
  final List<AiModelAdapter> _adapters = [];
  final Map<String, AiResult> _cache = {};
  static const Duration _cacheTtl = Duration(seconds: 300);
  int _currentAdapterIndex = 0;

  void registerAdapter(AiModelAdapter adapter) => _adapters.add(adapter);

  Future<AiResult> generateWithFallback(String prompt, {bool isVision = false, String? imageBase64}) async {
    // Check cache
    final cacheKey = '$prompt${imageBase64 ?? ''}';
    if (_cache.containsKey(cacheKey)) {
      final cached = _cache[cacheKey]!;
      return cached;
    }

    // Try each adapter with retry logic
    for (int i = 0; i < _adapters.length; i++) {
      final adapterIndex = (_currentAdapterIndex + i) % _adapters.length;
      final adapter = _adapters[adapterIndex];

      if (await adapter.isAvailable()) {
        for (int retry = 0; retry < 3; retry++) {
          try {
            AiResult result;
            if (isVision && imageBase64 != null) {
              result = await adapter.generateWithImage(prompt, imageBase64);
            } else {
              result = await adapter.generateText(prompt);
            }

            if (result.isSuccess) {
              _cache[cacheKey] = result;
              _currentAdapterIndex = adapterIndex;
              return result;
            }
          } catch (e) {
            // Exponential backoff
            await Future.delayed(Duration(milliseconds: 1000 * (retry + 1)));
          }
        }
      }
    }

    return AiResult(text: '', isSuccess: false, error: 'Todos los proveedores de IA fallaron');
  }

  void clearCache() => _cache.clear();
}
```

```dart
// lib/core/symmetry/symmetry_rank_system.dart
enum SymmetryRank {
  iron(0, 'Hierro', 0xFF6C757D),
  bronze(1, 'Bronce', 0xFFCD7F32),
  silver(2, 'Plata', 0xFFC0C0C0),
  gold(3, 'Oro', 0xFFFFD700),
  platinum(4, 'Platino', 0xFFE5E4E2),
  emerald(5, 'Esmeralda', 0xFF39FF14),
  diamond(6, 'Diamante', 0xFFB9F2FF),
  master(7, 'Maestro', 0xFF9D00FF),
  champion(8, 'Campeón', 0xFFFF0040),
  symmetric(9, 'Simétrico', 0xFFFFD700);

  final int level;
  final String displayName;
  final int colorValue;
  const SymmetryRank(this.level, this.displayName, this.colorValue);
}

class SymmetryRankSystem {
  static const List<int> _rankThresholds = [0, 100, 300, 600, 1000, 1500, 2500, 4000, 6000, 10000];

  SymmetryRank getRankFromXp(int xp) {
    for (int i = _rankThresholds.length - 1; i >= 0; i--) {
      if (xp >= _rankThresholds[i]) return SymmetryRank.values[i];
    }
    return SymmetryRank.iron;
  }

  int getXpForNextRank(int currentXp) {
    final currentRank = getRankFromXp(currentXp);
    if (currentRank == SymmetryRank.symmetric) return currentXp;
    return _rankThresholds[currentRank.level + 1];
  }

  double getProgressToNextRank(int currentXp) {
    final currentRank = getRankFromXp(currentXp);
    if (currentRank == SymmetryRank.symmetric) return 1.0;
    final currentThreshold = _rankThresholds[currentRank.level];
    final nextThreshold = _rankThresholds[currentRank.level + 1];
    return (currentXp - currentThreshold) / (nextThreshold - currentThreshold);
  }

  int calculateXpFromWorkout({required double tonnage, required int sets, required bool metProteinGoal}) {
    double xp = (tonnage / 100) + (sets * 2);
    if (metProteinGoal) xp *= 1.2; // Protein multiplier
    return xp.round();
  }
}
```

```dart
class LocaleNotifier extends StatefulWidget {

class _LocaleNotifierState extends State<LocaleNotifier> {
  late Locale _locale;
  static GlobalKey<NavigatorState> navigatorKey = GlobalKey();
  static _LocaleNotifierState? _instance;
  static _LocaleNotifierState? get instance => _instance;

  @override
  void initState() { super.initState(); _locale = widget.initialLocale; _instance = this; }

  static void updateLocale(Locale locale) => _instance?.setLocaleInternal(locale);

  void setLocaleInternal(Locale locale) async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString('language_code', locale.languageCode);
    if (mounted) setState(() => _locale = locale);

      create: (context) => FoodLogCubit(widget.foodRepository)..loadDailyLog(),

          navigatorKey: navigatorKey,
          debugShowCheckedModeBanner: false,
          title: 'CalAI',

            '/settings': (context) => const SettingsScreen(),
          },
        ),
      ),
    );
  }
```
